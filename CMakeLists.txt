cmake_minimum_required(VERSION 3.18)
project(BiWorldDB LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(BW_STATES 2049 CACHE STRING "Number of workload states")
set(BW_MAXQ 32 CACHE STRING "Maximum queries per state")
find_package(OpenMP REQUIRED)
find_package(OpenSSL REQUIRED)

function(worker target source)
  add_executable(${target} src/${source}.cpp)
  target_include_directories(${target} PRIVATE src)
  target_compile_definitions(${target} PRIVATE BWH_STATES=${BW_STATES} BWH_MAXQ=${BW_MAXQ})
  target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX OpenSSL::Crypto)
endfunction()
